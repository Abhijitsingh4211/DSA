1. ENTER THE VERTICES AND EDGES AND REPRESENT IN ADJACENCY MATRIX AND PRINT IT

QUESTION

Write a C program to enter the number of vertices and edges, create an undirected graph using an adjacency matrix, and print the matrix.

ANSWER

#include <stdio.h>

int main()
{
    int graph[20][20] = {0};
    int vertices, edges, i, j, u, v;

    printf("Enter number of vertices: ");
    scanf("%d", &vertices);

    if (vertices < 1 || vertices > 20)
    {
        printf("Enter vertices between 1 and 20.");
        return 0;
    }

    printf("Enter number of edges: ");
    scanf("%d", &edges);

    printf("Enter edges (u v), using vertices 0 to %d:\n", vertices - 1);

    for (i = 0; i < edges; i++)
    {
        scanf("%d%d", &u, &v);

        if (u >= 0 && u < vertices && v >= 0 && v < vertices)
        {
            graph[u][v] = 1;
            graph[v][u] = 1;
        }
        else
            printf("Invalid edge. Enter it again.\n"), i--;
    }

    printf("\nAdjacency Matrix:\n");
    for (i = 0; i < vertices; i++)
    {
        for (j = 0; j < vertices; j++)
            printf("%d ", graph[i][j]);
        printf("\n");
    }

    return 0;
}

OUTPUT

Enter number of vertices: 4
Enter number of edges: 4
Enter edges (u v), using vertices 0 to 3:
0 1
0 2
1 2
2 3

Adjacency Matrix:
0 1 1 0
1 0 1 0
1 1 0 1
0 0 1 0

Space complexity: O(V²). This example creates an undirected graph; for a directed graph, remove the assignment graph[v][u] = 1.

2. CREATE THE GRAPH AS AN ADJACENCY LIST WITH OPERATIONS SUCH AS INSERTION, DELETION

QUESTION

Write a C program to create an undirected graph using an adjacency list and perform edge insertion and deletion.

ANSWER

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int vertex;
    struct Node *next;
};

struct Node *adj[20];

void insertEdge(int u, int v)
{
    struct Node *newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = v;
    newNode->next = adj[u];
    adj[u] = newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = u;
    newNode->next = adj[v];
    adj[v] = newNode;
}

void deleteOne(int u, int v)
{
    struct Node *temp = adj[u], *prev = NULL;

    while (temp != NULL)
    {
        if (temp->vertex == v)
        {
            if (prev == NULL)
                adj[u] = temp->next;
            else
                prev->next = temp->next;

            free(temp);
            return;
        }
        prev = temp;
        temp = temp->next;
    }
}

void deleteEdge(int u, int v)
{
    deleteOne(u, v);
    deleteOne(v, u);
}

void display(int vertices)
{
    int i;
    struct Node *temp;

    printf("\nAdjacency List:\n");
    for (i = 0; i < vertices; i++)
    {
        printf("%d: ", i);
        temp = adj[i];

        while (temp != NULL)
        {
            printf("%d -> ", temp->vertex);
            temp = temp->next;
        }
        printf("NULL\n");
    }
}

int main()
{
    int vertices, choice, u, v, i;

    for (i = 0; i < 20; i++)
        adj[i] = NULL;

    printf("Enter number of vertices (1-20): ");
    scanf("%d", &vertices);

    if (vertices < 1 || vertices > 20)
    {
        printf("Invalid number of vertices.");
        return 0;
    }

    do
    {
        printf("\n1. Insert edge\n2. Delete edge\n3. Display\n4. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 1 || choice == 2)
        {
            printf("Enter vertices u and v: ");
            scanf("%d%d", &u, &v);

            if (u < 0 || u >= vertices || v < 0 || v >= vertices || u == v)
                printf("Invalid vertices or self-edge.\n");
            else if (choice == 1)
                insertEdge(u, v);
            else
                deleteEdge(u, v);
        }
        else if (choice == 3)
            display(vertices);
        else if (choice != 4)
            printf("Invalid choice.\n");

    } while (choice != 4);

    return 0;
}

OUTPUT

Enter number of vertices (1-20): 3

1. Insert edge
2. Delete edge
3. Display
4. Exit
Enter choice: 1
Enter vertices u and v: 0 1

Enter choice: 1
Enter vertices u and v: 1 2

Enter choice: 3

Adjacency List:
0: 1 -> NULL
1: 2 -> 0 -> NULL
2: 1 -> NULL

Enter choice: 2
Enter vertices u and v: 1 2

Enter choice: 3

Adjacency List:
0: 1 -> NULL
1: 0 -> NULL
2: NULL

Insertion and deletion are performed for undirected edges. Each edge is stored in both endpoint lists. This simple program assumes the user does not insert the same edge more than once.

3. TRAVERSE THE ADJACENCY MATRIX WITH DEPTH FIRST SEARCH

QUESTION

Write a C program to traverse a graph represented by an adjacency matrix using recursive Depth First Search (DFS).

ANSWER

#include <stdio.h>

int graph[20][20], visited[20], vertices;

void DFS(int vertex)
{
    int i;

    printf("%d ", vertex);
    visited[vertex] = 1;

    for (i = 0; i < vertices; i++)
    {
        if (graph[vertex][i] == 1 && visited[i] == 0)
            DFS(i);
    }
}

int main()
{
    int edges, i, u, v, start;

    printf("Enter number of vertices (1-20): ");
    scanf("%d", &vertices);

    if (vertices < 1 || vertices > 20)
        return 0;

    printf("Enter number of edges: ");
    scanf("%d", &edges);

    printf("Enter edges (u v), vertices numbered 0 to %d:\n", vertices - 1);
    for (i = 0; i < edges; i++)
    {
        scanf("%d%d", &u, &v);
        if (u >= 0 && u < vertices && v >= 0 && v < vertices)
        {
            graph[u][v] = 1;
            graph[v][u] = 1;
        }
        else
        {
            printf("Invalid edge.\n");
            i--;
        }
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    if (start < 0 || start >= vertices)
    {
        printf("Invalid starting vertex.");
        return 0;
    }

    printf("DFS Traversal: ");
    DFS(start);

    return 0;
}

OUTPUT

Enter number of vertices (1-20): 4
Enter number of edges: 4
Enter edges (u v), vertices numbered 0 to 3:
0 1
0 2
1 2
2 3
Enter starting vertex: 0
DFS Traversal: 0 1 2 3

Time complexity: O(V²) using an adjacency matrix, because each visited vertex scans its entire row.

5. TRAVERSE THE ADJACENCY MATRIX WITH BREADTH FIRST SEARCH

QUESTION

Write a C program to traverse a graph represented by an adjacency matrix using Breadth First Search (BFS).

ANSWER

#include <stdio.h>

int graph[20][20], visited[20], queue[20], vertices;

void BFS(int start)
{
    int front = 0, rear = 0, vertex, i;

    visited[start] = 1;
    queue[rear++] = start;

    while (front < rear)
    {
        vertex = queue[front++];
        printf("%d ", vertex);

        for (i = 0; i < vertices; i++)
        {
            if (graph[vertex][i] == 1 && visited[i] == 0)
            {
                visited[i] = 1;
                queue[rear++] = i;
            }
        }
    }
}
