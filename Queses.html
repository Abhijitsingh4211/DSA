QUESTION 1:-

IMPLEMENT THE OPERATIONS OF SIMPLE QUEUE IN AN INTEGER ARRAY (INSERT, DELETE, PEEK, ISEMPTY, ISFULL)


ANSWER:-
#include <stdio.h>
#define MAX 5
int q[MAX], front = -1, rear = -1;

int isEmpty() { return front == -1; }
int isFull() { return rear == MAX - 1; }

void insert(int x) {
    if (isFull()) { printf("Queue is full\n"); return; }
    if (isEmpty()) front = 0;
    q[++rear] = x;
}
void deleteItem() {
    if (isEmpty()) { printf("Queue is empty\n"); return; }
    printf("Deleted: %d\n", q[front]);
    if (front == rear) front = rear = -1;
    else front++;
}
void peek() {
    if (isEmpty()) printf("Queue is empty\n");
    else printf("Front: %d\n", q[front]);
}
int main() {
    insert(10); insert(20); insert(30);
    peek(); deleteItem(); peek();
    printf("isEmpty: %s\n", isEmpty() ? "Yes" : "No");
    printf("isFull: %s\n", isFull() ? "Yes" : "No");
    return 0;
}
OUTPUT:-
Front: 10
Deleted: 10
Front: 20
isEmpty: No
isFull: No

QUESTION 3:-

IMPLEMENT THE OPERATIONS OF QUEUE DYNAMICALLY OF INTEGERS (INSERT, DELETE, PEEK, ISEMPTY, ISFULL)


ANSWER:-
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int data; struct Node *next; } Node;
Node *front = NULL, *rear = NULL;
int isEmpty() { return front == NULL; }
int isFull() { return 0; } /* Linked queue has no fixed capacity; allocation can still fail. */
void insert(int x) {
    Node *n = (Node *)malloc(sizeof(Node));
    if (n == NULL) { printf("Memory allocation failed\n"); return; }
    n->data = x; n->next = NULL;
    if (rear == NULL) front = rear = n;
    else { rear->next = n; rear = n; }
}
void deleteItem() {
    if (isEmpty()) { printf("Queue is empty\n"); return; }
    Node *t = front; printf("Deleted: %d\n", t->data);
    front = front->next; if (front == NULL) rear = NULL; free(t);
}
void peek() {
    if (isEmpty()) printf("Queue is empty\n");
    else printf("Front: %d\n", front->data);
}
int main() {
    insert(10); insert(20); insert(30); peek(); deleteItem(); peek();
    printf("isEmpty: %s\n", isEmpty() ? "Yes" : "No");
    printf("isFull (fixed limit): %s\n", isFull() ? "Yes" : "No");
    while (front) deleteItem();
    return 0;
}
OUTPUT:-
Front: 10
Deleted: 10
Front: 20
isEmpty: No
isFull (fixed limit): No

QUESTION 7:-

IMPLEMENT THE OPERATIONS OF PRIORITY QUEUE IN AN INTEGER ARRAY (INSERT, DELETE, PEEK, ISEMPTY, ISFULL)


ANSWER:-
#include <stdio.h>
#define MAX 5
typedef struct { int value, priority; } Item;
Item q[MAX]; int count = 0;
int isEmpty() { return count == 0; }
int isFull() { return count == MAX; }
void insert(int value, int priority) {
    if (isFull()) { printf("Queue is full\n"); return; }
    int i = count - 1;
    while (i >= 0 && q[i].priority < priority) { q[i + 1] = q[i]; i--; }
    q[i + 1].value = value; q[i + 1].priority = priority; count++;
}
void deleteItem() {
    if (isEmpty()) { printf("Queue is empty\n"); return; }
    printf("Deleted: %d (priority %d)\n", q[0].value, q[0].priority);
    for (int i = 1; i < count; i++) q[i - 1] = q[i];
    count--;
}
void peek() {
    if (isEmpty()) printf("Queue is empty\n");
    else printf("Highest priority item: %d\n", q[0].value);
}
int main() {
    insert(100, 1); insert(200, 3); insert(300, 2);
    peek(); deleteItem(); peek();
    printf("isEmpty: %s\n", isEmpty() ? "Yes" : "No");
    printf("isFull: %s\n", isFull() ? "Yes" : "No");
    return 0;
}
OUTPUT:-
Highest priority item: 200
Deleted: 200 (priority 3)
Highest priority item: 300
isEmpty: No
isFull: No

IMPLEMENT THE OPERATIONS OF PRIORITY QUEUE DYNAMICALLY OF INTEGERS (INSERT, DELETE, PEEK, ISEMPTY, ISFULL)


ANSWER:-
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int value, priority; struct Node *next; } Node;
Node *front = NULL;
int isEmpty() { return front == NULL; }
int isFull() { return 0; }
void insert(int value, int priority) {
    Node *n = (Node *)malloc(sizeof(Node));
    if (!n) { printf("Memory allocation failed\n"); return; }
    n->value = value; n->priority = priority;
    if (!front || priority > front->priority) { n->next = front; front = n; }
    else {
        Node *p = front;
        while (p->next && p->next->priority >= priority) p = p->next;
        n->next = p->next; p->next = n;
    }
}
void deleteItem() {
    if (isEmpty()) { printf("Queue is empty\n"); return; }
    Node *t = front; printf("Deleted: %d (priority %d)\n", t->value, t->priority);
    front = front->next; free(t);
}
void peek() {
    if (isEmpty()) printf("Queue is empty\n");
    else printf("Highest priority item: %d\n", front->value);
}
int main() {
    insert(10, 1); insert(30, 3); insert(20, 2);
    peek(); deleteItem(); peek();
    printf("isEmpty: %s\n", isEmpty() ? "Yes" : "No");
    printf("isFull (fixed limit): %s\n", isFull() ? "Yes" : "No");
    while (front) deleteItem();
    return 0;
}

QUESTION 13:-

FIND THE FIRST CIRCULAR TOUR THAT VISITS ALL PETROL PUMPS


ANSWER:-
#include <stdio.h>
typedef struct { int petrol, distance; } Pump;
int circularTour(Pump p[], int n) {
    int start = 0, balance = 0, deficit = 0;
    for (int i = 0; i < n; i++) {
        balance += p[i].petrol - p[i].distance;
        if (balance < 0) { start = i + 1; deficit += balance; balance = 0; }
    }
    return (balance + deficit >= 0 && start < n) ? start : -1;
}
int main() {
    Pump p[] = {{4, 6}, {6, 5}, {7, 3}, {4, 5}};
    int n = sizeof(p) / sizeof(p[0]);
    int ans = circularTour(p, n);
    if (ans == -1) printf("No possible tour\n");
    else printf("First petrol pump index: %d\n", ans);
    return 0;
}
OUTPUT:-
First petrol pump index: 1
QUESTION 14:-

LENGTH OF THE LONGEST VALID SUBSTRING


ANSWER:-
#include <stdio.h>
#include <string.h>
#define MAX 100
int longestValid(char s[]) {
    int st[MAX], top = -1, last = -1, best = 0;
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '(') st[++top] = i;
        else {
            if (top == -1) last = i;
            else {
                top--;
                int len = (top == -1) ? i - last : i - st[top];
                if (len > best) best = len;
            }
        }
    }
    return best;
}
int main() {
    char s[] = ")()())";
    printf("String: %s\n", s);
    printf("Longest valid substring length: %d\n", longestValid(s));
    return 0;
}
OUTPUT:-
String: )()())
Longest valid substring length: 4
