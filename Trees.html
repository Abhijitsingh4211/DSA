QUESTION 1:-

CONSTRUCT BINARY TREE FROM PREORDER AND INORDER TRAVERSAL


ANSWER:-
#include <stdio.h>
#include <stdlib.h>
int pre[] = {1, 2, 4, 5, 3}, in[] = {4, 2, 5, 1, 3}, index = 0;
typedef struct Node { int data; struct Node *left, *right; } Node;
Node *make(int x) { Node *n = malloc(sizeof(Node)); n->data=x; n->left=n->right=NULL; return n; }
int find(int x, int l, int r) { for(int i=l;i<=r;i++) if(in[i]==x) return i; return -1; }
Node *build(int l,int r) {
    if(l>r) return NULL;
    int x=pre[index++], k=find(x,l,r); Node *n=make(x);
    n->left=build(l,k-1); n->right=build(k+1,r); return n;
}
void post(Node *r) { if(r){post(r->left);post(r->right);printf("%d ",r->data);} }
int main(){ Node *root=build(0,4); printf("Postorder: ");post(root);return 0; }
OUTPUT:-
Postorder: 4 5 2 3 1

int removeNode(int i,int x){
 if(i>=SIZE||t[i]==EMPTY)return i;

 else {int l=2*i+1,r=2*i+2;

  else {int m=minIndex(r),v=t[m];t[i]=v;removeNode(r,v);}
 }return i;
}

ANSWER:-
#include <stdio.h>
#include <stdlib.h>
typedef struct Node{int data;struct Node*l,*r;}Node;
Node*insert(Node*n,int x){if(!n){n=malloc(sizeof(Node));n->data=x;n->l=n->r=NULL;}else if(x<n->data)n->l=insert(n->l,x);else if(x>n->data)n->r=insert(n->r,x);return n;}
int search(Node*n,int x){if(!n)return 0;if(n->data==x)return 1;return x<n->data?search(n->l,x):search(n->r,x);}
Node*minimum(Node*n){while(n&&n->l)n=n->l;return n;}
Node*deleteNode(Node*n,int x){if(!n)return NULL;if(x<n->data)n->l=deleteNode(n->l,x);else if(x>n->data)n->r=deleteNode(n->r,x);else if(!n->l){Node*t=n->r;free(n);return t;}else if(!n->r){Node*t=n->l;free(n);return t;}else{Node*t=minimum(n->r);n->data=t->data;n->r=deleteNode(n->r,t->data);}return n;}
void in(Node*n){if(n){in(n->l);printf("%d ",n->data);in(n->r);}}
void pre(Node*n){if(n){printf("%d ",n->data);pre(n->l);pre(n->r);}}
void post(Node*n){if(n){post(n->l);post(n->r);printf("%d ",n->data);}}

FIND INORDER PREDECESSOR AND INORDER SUCCESSOR OF BINARY SEARCH TREE (DYNAMICALLY REPRESENTED)

#include <stdlib.h>

Node*ins(Node*n,int x){if(!n){n=malloc(sizeof(Node));n->data=x;n->l=n->r=NULL;}else if(x<n->data)n->l=ins(n->l,x);else n->r=ins(n->r,x);return n;}
void findPredSucc(Node*n,int key,Node**pre,Node**suc){
 *pre=*suc=NULL;
 while(n){if(n->data==key){if(n->l){Node*t=n->l;while(t->r)t=t->r;*pre=t;}if(n->r){Node*t=n->r;while(t->l)t=t->l;*suc=t;}break;}
 if(key<n->data){*suc=n;n=n->l;}else{*pre=n;n=n->r;}}
}
