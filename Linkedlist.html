LINKED LIST PROGRAMS
QUESTION 1:-

IMPLEMENT THE OPERATIONS OF SINGLE LINKED LIST STORING INTEGER VALUES (INSERT AT BEGIN, INSERT AT POS, INSERT AT END, DELETE AT BEGIN, DELETE AT END, DELETE AT POS, ISEMPTY, ISFULL, DISPLAY, SEARCH)


ANSWER:-
#include <stdio.h>
#include <stdlib.h>
#define CIRCULAR 0
#define DOUBLY 0
struct Node {
    int data;
    struct Node *prev;
    struct Node *next;
};
struct Node *head=NULL;
struct Node *newNode(int value) {
    struct Node *n=(struct Node*)malloc(sizeof(struct Node));
    if(!n) {
        printf("Memory allocation failed\n");
        exit(1);
    }
    n->data=value;
    n->prev=NULL;
    n->next=NULL;
    return n;
}
void linearize(void) {
    struct Node *p;
    if(CIRCULAR && head) {
        p=head;
        while(p->next!=head) p=p->next;
        p->next=NULL;
    }
}
void repair(void) {
    struct Node *p=head,*prev=NULL;
    while(p) {
        if(DOUBLY) p->prev=prev;
        prev=p;
        p=p->next;
    }
    if(CIRCULAR && head) {
        prev->next=head;
        if(DOUBLY) head->prev=prev;
    }
}
int isEmpty(void) {
    return head==NULL;
}
int isFull(void) {
    struct Node *p=(struct Node*)malloc(sizeof(struct Node));
    if(!p) return 1;
    free(p);
    return 0;
}
void insertBegin(int value) {
    struct Node *n=newNode(value);
    linearize();
    n->next=head;
    head=n;
    repair();
}
void insertEnd(int value) {
    struct Node *n=newNode(value),*p;
    linearize();
    if(!head) head=n;
    else {
        p=head;
        while(p->next) p=p->next;
        p->next=n;
    }
    repair();
}
void insertPos(int value,int pos) {
    int i;
    struct Node *p,*n;
    if(pos<=1 || !head) {
        insertBegin(value);
        return;
    }
    linearize();
    p=head;
    for(i=1;i<pos-1 && p->next;i++) p=p->next;
    n=newNode(value);
    n->next=p->next;
    p->next=n;
    repair();
}
void deleteBegin(void) {
    struct Node *p;
    linearize();
    if(!head) {
        printf("List is empty\n");
        return;
    }
    p=head;
    head=head->next;
    free(p);
    repair();
}
void deleteEnd(void) {
    struct Node *p,*prev=NULL;
    linearize();
    if(!head) {
        printf("List is empty\n");
        return;
    }
    p=head;
    while(p->next) {
        prev=p;
        p=p->next;
    }
    if(prev) prev->next=NULL;
    else head=NULL;
    free(p);
    repair();
}
void deletePos(int pos) {
    int i;
    struct Node *p,*prev=NULL;
    linearize();
    if(!head) {
        printf("List is empty\n");
        return;
    }
    if(pos<=1) {
        p=head;
        head=head->next;
        free(p);
        repair();
        return;
    }
    p=head;
    for(i=1;i<pos && p;i++) {
        prev=p;
        p=p->next;
    }
    if(!p) {
        printf("Invalid position\n");
        repair();
        return;
    }
    prev->next=p->next;
    free(p);
    repair();
}
void display(void) {
    struct Node *p=head;
    if(!p) {
        printf("List is empty\n");
        return;
    }
    do {
        printf("%d ",p->data);
        p=p->next;
    }
    while(p && (!CIRCULAR || p!=head));
    printf("\n");
}
int search(int value) {
    int pos=1;
    struct Node *p=head;
    if(!p) return -1;
    do {
        if(p->data==value) return pos;
        p=p->next;
        pos++;
    }
    while(p && (!CIRCULAR || p!=head));
    return -1;
}
int main(void) {
    int choice,pos,found;
    int value;
    do {
        printf("\n1.Insert at begin\n2.Insert at position\n3.Insert at end\n4.Delete at begin\n5.Delete at end\n6.Delete at position\n7.IsEmpty\n8.IsFull\n9.Display\n10.Search\n0.Exit\nEnter choice: ");
        if(scanf("%d",&choice)!=1) break;
        switch(choice) {
            case 1: printf("Enter value: ");
            scanf("%d",&value);
            insertBegin(value);
            break;
            case 2: printf("Enter value and position: ");
            scanf("%d %d",&value,&pos);
            insertPos(value,pos);
            break;
            case 3: printf("Enter value: ");
            scanf("%d",&value);
            insertEnd(value);
            break;
            case 4: deleteBegin();
            break;
            case 5: deleteEnd();
            break;
            case 6: printf("Enter position: ");
            scanf("%d",&pos);
            deletePos(pos);
            break;
            case 7: printf(isEmpty()?"List is empty\n":"List is not empty\n");
            break;
            case 8: printf(isFull()?"Memory allocation unavailable\n":"Memory available\n");
            break;
            case 9: display();
            break;
            case 10: printf("Enter value to search: ");
            scanf("%d",&value);
            found=search(value);
            if(found<0) printf("Value not found\n");
            else printf("Value found at position %d\n",found);
            break;
            case 0: break;
            default: printf("Invalid choice\n");
        }
    }
    while(choice!=0);

QUESTION 9:-

ENTER THE 2 POLYNOMIAL EQUATIONS DYNAMICALLY, ADD THEM AND DISPLAY THE ANSWER.


ANSWER:-
#include <stdio.h>
#include <stdlib.h>
struct Term {
    int coeff, exp;
    struct Term *next;
};
struct Term *createTerm(int c,int e) {
    struct Term *n=(struct Term*)malloc(sizeof(struct Term));
    if(!n) exit(1);
    n->coeff=c;
    n->exp=e;
    n->next=NULL;
    return n;
}
void insertTerm(struct Term **head,int c,int e) {
    struct Term *p=*head,*prev=NULL;
    while(p && p->exp>e) {
        prev=p;
        p=p->next;
    }
    if(p && p->exp==e) {
        p->coeff+=c;
        return;
    }
    {
        struct Term *n=createTerm(c,e);
        n->next=p;
        if(prev) prev->next=n;
        else *head=n;
    }
}
void readPoly(struct Term **p) {
    int n,i,c,e;
    printf("Enter number of terms: ");
    scanf("%d",&n);
    for(i=0;i<n;i++) {
        printf("Enter coefficient and exponent: ");
        scanf("%d%d",&c,&e);
        insertTerm(p,c,e);
    }
}
void display(struct Term *p) {
    int first=1;
    if(!p) {
        printf("0");
        return;
    }
    while(p) {
        if(p->coeff) {
            if(!first && p->coeff>0) printf(" + ");
            if(p->coeff<0) printf(" - ");
            printf("%d",abs(p->coeff));
            if(p->exp>0) printf("x");
            if(p->exp>1) printf("^%d",p->exp);
            first=0;
        }
        p=p->next;
    }
    if(first) printf("0");
}

QUESTION 10:-

ENTER THE 2 POLYNOMIAL EQUATIONS DYNAMICALLY, SUBTRACT THEM AND DISPLAY THE ANSWER.


ANSWER:-
#include <stdio.h>
#include <stdlib.h>
struct Term {
    int coeff, exp;
    struct Term *next;
};
struct Term *createTerm(int c,int e) {
    struct Term *n=(struct Term*)malloc(sizeof(struct Term));
    if(!n) exit(1);
    n->coeff=c;
    n->exp=e;
    n->next=NULL;
    return n;
}
void insertTerm(struct Term **head,int c,int e) {
    struct Term *p=*head,*prev=NULL;
    while(p && p->exp>e) {
        prev=p;
        p=p->next;
    }
    if(p && p->exp==e) {
        p->coeff+=c;
        return;
    }
    {
        struct Term *n=createTerm(c,e);
        n->next=p;
        if(prev) prev->next=n;
        else *head=n;
    }
}
void readPoly(struct Term **p) {
    int n,i,c,e;
    printf("Enter number of terms: ");
    scanf("%d",&n);
    for(i=0;i<n;i++) {
        printf("Enter coefficient and exponent: ");
        scanf("%d%d",&c,&e);
        insertTerm(p,c,e);
    }
}
void display(struct Term *p) {
    int first=1;
    if(!p) {
        printf("0");
        return;
    }
    while(p) {
        if(p->coeff) {
            if(!first && p->coeff>0) printf(" + ");
            if(p->coeff<0) printf(" - ");
            printf("%d",abs(p->coeff));
            if(p->exp>0) printf("x");
            if(p->exp>1) printf("^%d",p->exp);
            first=0;
        }
        p=p->next;
    }
    if(first) printf("0");
}
