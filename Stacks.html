QUESTION 1:-

CREATE A STACK IMPLEMENTING THE OPERATIONS ON AN INTEGER ARRAY (PUSH, POP, PEEK, ISEMPTY, ISFULL)


ANSWER:-
#include <stdio.h>
#define MAX 5

int stack[MAX], top = -1;

void push(int value) {
    if(top == MAX - 1)
        printf("Stack is full.\n");
    else {
        top++;
        stack[top] = value;
        printf("%d pushed into stack.\n", value);
    }
}

void pop() {
    if(top == -1)
        printf("Stack is empty.\n");
    else {
        printf("Popped element: %d\n", stack[top]);
        top--;
    }
}

void peek() {
    if(top == -1)
        printf("Stack is empty.\n");
    else
        printf("Top element: %d\n", stack[top]);
}

int isEmpty() {
    return top == -1;
}

int isFull() {
    return top == MAX - 1;
}

int main() {
    int choice, value;

    do {
        printf("\n1. PUSH\n2. POP\n3. PEEK\n4. ISEMPTY\n5. ISFULL\n6. EXIT\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice) {
            case 1:
                printf("Enter value: ");
                scanf("%d", &value);
                push(value);
                break;
            case 2:
                pop();
                break;
            case 3:
                peek();
                break;
            case 4:
                if(isEmpty())
                    printf("Stack is empty.\n");
                else
                    printf("Stack is not empty.\n");
                break;
            case 5:
                if(isFull())
                    printf("Stack is full.\n");
                else
                    printf("Stack is not full.\n");
                break;
            case 6:
                printf("Program ended.\n");
                break;
            default:
                printf("Invalid choice.\n");
        }
    } while(choice != 6);

    return 0;
}

OUTPUT:-

2. POP
3. PEEK
4. ISEMPTY
5. ISFULL
6. EXIT

CREATE A STACK IMPLEMENTING THE OPERATIONS ON THE INTEGER NODES (DYNAMICALLY) (PUSH, POP, PEEK, ISEMPTY, ISFULL)

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

struct node *top = NULL;

void push(int value) {
    struct node *newnode;
    newnode = (struct node *)malloc(sizeof(struct node));

    if(newnode == NULL) {
        printf("Memory is full.\n");
        return;
    }

    newnode->data = value;
    newnode->next = top;
    top = newnode;
    printf("%d pushed into stack.\n", value);
}

void pop() {
    struct node *temp;

    if(top == NULL)
        printf("Stack is empty.\n");
    else {
        temp = top;
        printf("Popped element: %d\n", temp->data);
        top = top->next;
        free(temp);
    }
}

void peek() {
    if(top == NULL)
        printf("Stack is empty.\n");
    else
        printf("Top element: %d\n", top->data);
}

int isEmpty() {
    return top == NULL;
}

int isFull() {
    struct node *temp = (struct node *)malloc(sizeof(struct node));
    if(temp == NULL)
        return 1;
    free(temp);
    return 0;
}

    int choice, value;

    do {
        printf("\n1. PUSH\n2. POP\n3. PEEK\n4. ISEMPTY\n5. ISFULL\n6. EXIT\n");

                break;
            case 6:
                printf("Program ended.\n");
                break;
            default:
                printf("Invalid choice.\n");
        }
    } while(choice != 6);

        pop();

ENTER THE STRING FROM THE USER AND REVERSE THE STRING USING STACKS.


ANSWER:-
#include <stdio.h>
#include <string.h>
#define MAX 100

int main() {
    char str[MAX], stack[MAX];
    int top = -1, i;

    printf("Enter a string: ");
    fgets(str, MAX, stdin);
    str[strcspn(str, "\n")] = '\0';

    for(i = 0; str[i] != '\0'; i++)
        stack[++top] = str[i];

    while(top != -1)
        printf("%c", stack[top--]);

QUESTION 6:-

CREATE THE PROGRAM TO MATCH THE SINGLE BRACKET USING STACKS.


ANSWER:-
#include <stdio.h>
#include <string.h>
#define MAX 100

int main() {
    char str[MAX], stack[MAX];
    int top = -1, i, valid = 1;

    printf("Enter an expression: ");
    fgets(str, MAX, stdin);

    for(i = 0; str[i] != '\0'; i++) {
        if(str[i] == '(' || str[i] == '[' || str[i] == '{')
            stack[++top] = str[i];
        else if(str[i] == ')' || str[i] == ']' || str[i] == '}') {
            if(top == -1) {
                valid = 0;
                break;
            }

            if((str[i] == ')' && stack[top] == '(') ||
               (str[i] == ']' && stack[top] == '[') ||
               (str[i] == '}' && stack[top] == '{'))
                top--;
            else {
                valid = 0;
                break;
            }
        }
    }

    if(top != -1)
        valid = 0;

    if(valid)
        printf("Brackets are balanced.\n");
    else
        printf("Brackets are not balanced.\n");

    return 0;
}

Enter an expression: (A+B)*[C-D]
Brackets are balanced.

QUESTION 8:-

SORT THE STACK IN ASCENDING ORDER USING INTEGER ARRAY.


ANSWER:-
#include <stdio.h>
#define MAX 100

int main() {
    int stack[MAX], tempStack[MAX];
    int top = -1, tempTop = -1;
    int n, i, value;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if(n < 1 || n > MAX) {
        printf("Invalid size.\n");
        return 0;
    }

    for(i = 0; i < n; i++) {
        printf("Enter element: ");
        scanf("%d", &stack[++top]);
    }

    while(top != -1) {
        value = stack[top--];

        while(tempTop != -1 && tempStack[tempTop] < value)
            stack[++top] = tempStack[tempTop--];

        tempStack[++tempTop] = value;
    }

    while(tempTop != -1)
        stack[++top] = tempStack[tempTop--];

    printf("Stack in ascending order (bottom to top): ");
    for(i = 0; i <= top; i++)
        printf("%d ", stack[i]);

    printf("\n");
    return 0;
}

Enter element: 4
Enter element: 1
Enter element: 3

QUESTION 9:-

ENTER THE INFIX EXPRESSION FROM USER AND CONVERT INTO PREFIX EXPRESSION.


ANSWER:-
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 100

int precedence(char ch) {
    if(ch == '^') return 3;
    if(ch == '*' || ch == '/') return 2;
    if(ch == '+' || ch == '-') return 1;
    return 0;
}

int main() {
    char infix[MAX], stack[MAX], prefix[MAX], temp;
    int top = -1, i, j = 0;

    printf("Enter infix expression: ");
    scanf("%99s", infix);

    for(i = 0; infix[i] != '\0'; i++) {
        if(infix[i] == '(') infix[i] = ')';
        else if(infix[i] == ')') infix[i] = '(';
    }

    for(i = 0, j = (int)strlen(infix) - 1; i < j; i++, j--) {
        temp = infix[i];
        infix[i] = infix[j];
        infix[j] = temp;
    }

    j = 0;
    for(i = 0; infix[i] != '\0'; i++) {
        if(isalnum((unsigned char)infix[i]))
            prefix[j++] = infix[i];
        else if(infix[i] == '(')
            stack[++top] = infix[i];
        else if(infix[i] == ')') {
            while(top != -1 && stack[top] != '(')
                prefix[j++] = stack[top--];
            if(top != -1) top--;
        }
        else {
            while(top != -1 && stack[top] != '(' &&
                  (precedence(stack[top]) > precedence(infix[i]) ||
                  (precedence(stack[top]) == precedence(infix[i]) && infix[i] == '^')))
                prefix[j++] = stack[top--];
            stack[++top] = infix[i];
        }
    }

    while(top != -1)
        prefix[j++] = stack[top--];

    prefix[j] = '\0';

    for(i = 0, j = (int)strlen(prefix) - 1; i < j; i++, j--) {
        temp = prefix[i];
        prefix[i] = prefix[j];
        prefix[j] = temp;
    }

    printf("Prefix expression: %s\n", prefix);
    return 0;
}

Prefix expression: +A*BC

QUESTION 11:-

CREATE A PROGRAM TO EVALUATE THE PREFIX EXPRESSION.


ANSWER:-
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 100

int power(int a, int b) {
    int result = 1, i;
    for(i = 0; i < b; i++)
        result = result * a;
    return result;
}

int calculate(int a, int b, char op) {
    switch(op) {
        case '+': return a + b;
        case '-': return a - b;
        case '*': return a * b;
        case '/': return a / b;
        case '^': return power(a, b);
    }
    return 0;
}

int main() {
    char exp[MAX];
    int stack[MAX], top = -1, i, a, b;

    printf("Enter prefix expression: ");
    scanf("%99s", exp);

    for(i = (int)strlen(exp) - 1; i >= 0; i--) {
        if(isdigit((unsigned char)exp[i]))
            stack[++top] = exp[i] - '0';
        else {
            if(top < 1) {
                printf("Invalid expression.\n");
                return 0;
            }
            a = stack[top--];
            b = stack[top--];

            if(exp[i] == '/' && b == 0) {
                printf("Division by zero is not allowed.\n");
                return 0;
            }

            stack[++top] = calculate(a, b, exp[i]);
        }
    }

    if(top == 0)
        printf("Result: %d\n", stack[top]);
    else
        printf("Invalid expression.\n");

    return 0;
}

OUTPUT:-
Enter prefix expression: +9*26
Result: 21

QUESTION 13:-

IMPLEMENT THE TOWER OF HANOI USING STACKS.


ANSWER:-
#include <stdio.h>
#define MAX 20

struct Stack {
    int disk[MAX];
    int top;
};

void push(struct Stack *s, int disk) {
    s->disk[++s->top] = disk;
}

int pop(struct Stack *s) {
    return s->disk[s->top--];
}

void moveDisks(struct Stack *a, struct Stack *b, char from, char to) {
    int diskA, diskB;

    if(a->top == -1) {
        diskB = pop(b);
        push(a, diskB);
        printf("Move disk %d from %c to %c\n", diskB, to, from);
    }
    else if(b->top == -1) {
        diskA = pop(a);
        push(b, diskA);
        printf("Move disk %d from %c to %c\n", diskA, from, to);
    }
    else if(a->disk[a->top] < b->disk[b->top]) {
        diskA = pop(a);
        push(b, diskA);
        printf("Move disk %d from %c to %c\n", diskA, from, to);
    }
    else {
        diskB = pop(b);
        push(a, diskB);
        printf("Move disk %d from %c to %c\n", diskB, to, from);
    }
}

int main() {
    struct Stack source, auxiliary, destination, temp;
    int n, i, totalMoves;
    char sourceName = 'A', auxiliaryName = 'B', destinationName = 'C';

    source.top = auxiliary.top = destination.top = -1;

    printf("Enter number of disks (1-%d): ", MAX);
    scanf("%d", &n);

    if(n < 1 || n > MAX) {
        printf("Invalid number of disks.\n");
        return 0;
    }

    for(i = n; i >= 1; i--)
        push(&source, i);

    totalMoves = (1 << n) - 1;

    if(n % 2 == 0) {
        temp = auxiliary;
        auxiliary = destination;
        destination = temp;

        auxiliaryName = 'C';
        destinationName = 'B';
    }

    for(i = 1; i <= totalMoves; i++) {
        if(i % 3 == 1)
            moveDisks(&source, &destination, sourceName, destinationName);
        else if(i % 3 == 2)
            moveDisks(&source, &auxiliary, sourceName, auxiliaryName);
        else
            moveDisks(&auxiliary, &destination, auxiliaryName, destinationName);
    }

    return 0;
}

OUTPUT:-
Enter number of disks (1-20): 3
Move disk 1 from A to C
Move disk 2 from A to B
Move disk 1 from C to B
Move disk 3 from A to C
Move disk 1 from B to A
Move disk 2 from B to C
Move disk 1 from A to C
