QUESTION 1:-

Enter a string from the user and convert it into uppercase.


ANSWER:-
#include <stdio.h>
#include <ctype.h>

int main() {
    char s[200];
    int i;
    printf("Enter a string: ");
    fgets(s, sizeof(s), stdin);
    for(i = 0; s[i] != '\0'; i++)
        s[i] = toupper((unsigned char)s[i]);

    return 0;

}

QUESTION 21:-

Print all permutations of the string ABCDE.


ANSWER:-
#include <stdio.h>
#include <string.h>

void permute(char s[], int left, int right) {
    int i;
    char temp;
    if(left == right) {
        printf("%s\n", s);
        return;
    }
    for(i = left; i <= right; i++) {
        temp = s[left]; s[left] = s[i]; s[i] = temp;
        permute(s, left + 1, right);
        temp = s[left]; s[left] = s[i]; s[i] = temp;
    }
}

int main() {
    char s[] = "ABCDE";
    permute(s, 0, (int)strlen(s) - 1);
    return 0;
}
OUTPUT:-
ABCDE\nABCED\nABDCE\nABDEC\nABECD\nABEDC\n...\nTotal permutations: 120

QUESTION 25:-

#include <ctype.h>

        end--;

        start++;

    return 0;
}

QUESTION 28:-

Create a dictionary of 10 words and meanings with create, delete and search operations.


ANSWER:-
#include <stdio.h>
#include <string.h>

struct Entry {
    char word[30];
    char meaning[100];
};

int main() {
    struct Entry d[20] = {
        {"Apple", "A fruit"}, {"Book", "A set of pages"},
        {"Computer", "An electronic device"}, {"Dog", "An animal"},
        {"Earth", "Our planet"}, {"Flower", "A plant part"},
        {"Garden", "A place for plants"}, {"House", "A building"},
        {"Ink", "A writing liquid"}, {"Joy", "Happiness"}
    };
    int n = 10, choice, i, pos;
    char word[30], meaning[100];
    do {
        printf("\n1.Create 2.Delete 3.Search 4.Display 5.Exit\nChoice: ");
        scanf("%d", &choice);
        if(choice == 1) {
            if(n == 10) { printf("Dictionary is full"); continue; }
            printf("Enter word: "); scanf("%29s", word);
            printf("Enter meaning: "); scanf(" %99[^\n]", meaning);
            strcpy(d[n].word, word); strcpy(d[n].meaning, meaning); n++;
        } else if(choice == 2 || choice == 3) {
            printf("Enter word: "); scanf("%29s", word);
            pos = -1;
            for(i = 0; i < n; i++)
                if(strcmp(d[i].word, word) == 0) { pos = i; break; }
            if(pos == -1) printf("Word not found");
            else if(choice == 3) printf("%s: %s", d[pos].word, d[pos].meaning);
            else {
                for(i = pos; i < n - 1; i++) d[i] = d[i + 1];
                n--;
                printf("Word deleted");
            }
        } else if(choice == 4) {
            for(i = 0; i < n; i++)
                printf("%s: %s\n", d[i].word, d[i].meaning);
        }
    } while(choice != 5);
    return 0;
}
