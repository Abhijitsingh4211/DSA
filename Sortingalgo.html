#include <stdio.h>

INTEGER ARRAY - MERGE SORT THROUGH RECURSION


ANSWER:-
#include <stdio.h>
int a[100], temp[100];
void merge(int l,int m,int r) {
    int i=l,j=m+1,k=l;
    while(i<=m && j<=r) temp[k++]=(a[i]<=a[j])?a[i++]:a[j++];
    while(i<=m) temp[k++]=a[i++];
    while(j<=r) temp[k++]=a[j++];
    for(i=l;i<=r;i++) a[i]=temp[i];
}
void mergeSort(int l,int r) {
    if(l>=r)return;
    int m=(l+r)/2;
    mergeSort(l,m);mergeSort(m+1,r);merge(l,m,r);
}
int main() {
    int n,i;printf("Enter number of elements: ");scanf("%d",&n);
    printf("Enter elements: ");for(i=0;i<n;i++)scanf("%d",&a[i]);
    mergeSort(0,n-1);
    printf("Sorted array: ");for(i=0;i<n;i++)printf("%d ",a[i]);
    return 0;
}
OUTPUT:-
Enter number of elements: 6
Enter elements: 38 27 43 3 9 82
Sorted array: 3 9 27 38 43 82
BEST AND WORST COMPLEXITY:-
Best and worst time: Θ(n log₂ n). Merge comparisons vary with data; per merge they are at most the number of elements merged minus one. Auxiliary array: O(n); recursion stack: O(log n).

int main() {

    for(w=1;w<n;w*=2) {
        for(l=0;l<n;l+=2*w) {
            m=l+w-1;r=l+2*w-1;if(m>=n)m=n-1;if(r>=n)r=n-1;
            x=l;y=m+1;k=l;

#include <stdio.h>
void quick(int a[],int low,int high) {
    if(low>=high)return;
    int pivot=a[high],i=low-1,j,temp;
    for(j=low;j<high;j++)if(a[j]<=pivot){i++;temp=a[i];a[i]=a[j];a[j]=temp;}
    temp=a[i+1];a[i+1]=a[high];a[high]=temp;
    int p=i+1;quick(a,low,p-1);quick(a,p+1,high);
}

    int low[100],high[100],top=-1;
    if(n<=1)return;low[++top]=0;high[top]=n-1;
    while(top>=0) {
        int l=low[top],h=high[top--],pivot=a[h],i=l-1,j,temp;
        for(j=l;j<h;j++)if(a[j]<=pivot){i++;temp=a[i];a[i]=a[j];a[j]=temp;}

    }
}
