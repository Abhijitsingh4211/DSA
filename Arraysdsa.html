QUESTION 9:-

Enter an array from user and find the k-th smallest value?


ANSWER:-
#include <stdio.h>

int main() {
    int a[100], n, i, j, temp, k;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    printf("Enter k: ");
    scanf("%d", &k);

    if(k < 1 || k > n) {
        printf("Invalid value of k");
        return 0;
    }

    for(i = 0; i < n - 1; i++) {
        for(j = i + 1; j < n; j++) {
            if(a[i] > a[j]) {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("%d-th smallest value: %d", k, a[k - 1]);

    return 0;
}
OUTPUT:-
Enter size of array: 5
Enter array elements: 8 3 6 1 5
Enter k: 3
3-th smallest value: 5

ANSWER:-
#include <stdio.h>

int main() {
    int a[20], n, i, mask, total, j;

    printf("Enter size of array (maximum 20): ");
    scanf("%d", &n);

    if(n < 1 || n > 20) {
        printf("Invalid array size");
        return 0;
    }

    printf("Enter array elements: ");
    for(i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    total = 1 << n;
    printf("All possible non-empty combinations:\n");

    for(mask = 1; mask < total; mask++) {
        for(j = 0; j < n; j++) {
            if(mask & (1 << j)) {
                printf("%d ", a[j]);
            }
        }
        printf("\n");
    }

    return 0;
}

#include <stdio.h>

int isPalindrome(int num) {
    int original = num, reverse = 0, digit;

    if(num < 0)
        return 0;

    while(num > 0) {
        digit = num % 10;
        reverse = reverse * 10 + digit;
        num = num / 10;
    }

    return original == reverse;
}

int main() {
    int a[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    printf("Palindrome numbers: ");
    for(i = 0; i < n; i++) {
        if(isPalindrome(a[i])) {
            printf("%d ", a[i]);
        }
    }

    return 0;
}

QUESTION 17:-

Merge the two sorted arrays into one array but in sorted order?


ANSWER:-
#include <stdio.h>

int main() {
    int a[100], b[100], c[200], n, m, i, j, k;

    printf("Enter size of first sorted array: ");
    scanf("%d", &n);

    printf("Enter first array elements in sorted order: ");
    for(i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    printf("Enter size of second sorted array: ");
    scanf("%d", &m);

    printf("Enter second array elements in sorted order: ");
    for(j = 0; j < m; j++) {
        scanf("%d", &b[j]);
    }

    i = 0;
    j = 0;
    k = 0;

    while(i < n && j < m) {
        if(a[i] <= b[j])
            c[k++] = a[i++];
        else
            c[k++] = b[j++];
    }

    while(i < n)
        c[k++] = a[i++];

    while(j < m)
        c[k++] = b[j++];

    printf("Merged sorted array: ");
    for(i = 0; i < k; i++) {
        printf("%d ", c[i]);
    }

    return 0;
}
OUTPUT:-
Enter size of first sorted array: 3
Enter first array elements in sorted order: 1 4 7
Enter size of second sorted array: 3
Enter second array elements in sorted order: 2 3 8
Merged sorted array: 1 2 3 4 7 8
QUESTION 18:-

Find the local minima in the array? (An element in an array is a local minima if it less than the element after it, and the element before it)


ANSWER:-
#include <stdio.h>

int main() {
    int a[100], n, i, found = 0;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    printf("Local minima: ");
    for(i = 1; i < n - 1; i++) {
        if(a[i] < a[i - 1] && a[i] < a[i + 1]) {
            printf("%d ", a[i]);
            found = 1;
        }
    }

    if(found == 0)
        printf("No local minima found");

    return 0;
}

QUESTION 20:-

Trapping Rain Water Problem: Given n non-negative integers representing an elevation map where the width of each bar is 1, compute how much water it can trap after raining.


ANSWER:-
#include <stdio.h>

int main() {
    int height[100], n, left, right, leftMax = 0, rightMax = 0;
    int water = 0;

    printf("Enter number of bars: ");
    scanf("%d", &n);

    printf("Enter bar heights: ");
    for(int i = 0; i < n; i++) {
        scanf("%d", &height[i]);
    }

    left = 0;
    right = n - 1;

    while(left < right) {
        if(height[left] < height[right]) {
            if(height[left] >= leftMax)
                leftMax = height[left];
            else
                water += leftMax - height[left];

            left++;
        } else {
            if(height[right] >= rightMax)
                rightMax = height[right];
            else
                water += rightMax - height[right];

            right--;
        }
    }

    printf("Trapped rain water: %d", water);

    return 0;
}
OUTPUT:-
Enter number of bars: 6
Enter bar heights: 0 1 0 2 1 0
Trapped rain water: 1
